rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return signedIn()
        &&
        get(
          /databases/$(database)/documents/users/$(request.auth.uid)
        ).data.role == "admin";
    }

    function isOwner(uid) {
      return signedIn()
        &&
        request.auth.uid == uid;
    }
    
    match /customerNotes/{noteId} {

allow read:
if isAdmin();

allow create:
if isAdmin();

allow update,
delete:
if false;

}

    match /users/{userId} {

      allow create:
        if signedIn()
        &&
        request.auth.uid == userId;

      allow read:
        if isOwner(userId)
        ||
        isAdmin();

      allow update:
        if isOwner(userId)
        ||
        isAdmin();

      allow delete:
        if isAdmin();
    }

    match /products/{productId} {

      allow read:
        if true;

      allow create,
            update,
            delete:
        if isAdmin();
    }

    match /orders/{orderId} {

      allow create:
        if signedIn();

      allow read:
        if isAdmin()
        ||
        resource.data.uid
        ==
        request.auth.uid;

      allow update:
        if isAdmin();

      allow delete:
        if isAdmin();
    }

    match /claims/{claimId} {

      allow create:
        if signedIn();

      allow read:
        if isAdmin()
        ||
        resource.data.uid
        ==
        request.auth.uid;

      allow update:
        if isAdmin();

      allow delete:
        if isAdmin();
    }

    match /claimNotes/{noteId} {

  allow create:
    if isAdmin();

  allow read:
    if isAdmin();

  allow update:
    if false;

  allow delete:
    if isAdmin();

}

match /claimTimeline/{timelineId} {

  allow create:
    if isAdmin();

  allow read:
    if isAdmin();

  allow update,
        delete:
    if false;

}

    match /inventory_logs/{logId} {

      allow read:
        if isAdmin();

      allow create:
        if isAdmin();

      allow update,
            delete:
        if false;
    }

    match /coupons/{couponId} {

      allow read:
        if true;

      allow create,
            update,
            delete:
        if isAdmin();
    }

    match /notifications/{notificationId} {

      allow read:
        if signedIn();

      allow create,
            update,
            delete:
        if isAdmin();
    }

    match /support/{ticketId} {

      allow create:
        if signedIn();

      allow read:
        if isAdmin()
        ||
        resource.data.uid
        ==
        request.auth.uid;

      allow update:
        if isAdmin();

      allow delete:
        if isAdmin();
    }

    match /banners/{bannerId} {

      allow read:
        if true;

      allow create,
            update,
            delete:
        if isAdmin();
    }

    match /activity/{activityId} {

      allow read:
        if isAdmin();

      allow create:
        if isAdmin();

      allow update,
            delete:
        if false;
    }

    match /settings/{settingId} {

  allow read:
    if true;

  allow create,
        update,
        delete:
    if isAdmin();

}

match /carts/{cartId} {

  allow create:
    if signedIn();

  allow read:
    if signedIn() &&
       resource.data.uid == request.auth.uid;

  allow update:
    if signedIn() &&
       resource.data.uid == request.auth.uid;

  allow delete:
    if signedIn() &&
       resource.data.uid == request.auth.uid;

}

match /wishlists/{wishlistId} {

  allow create:
    if signedIn();

  allow read:
    if signedIn() &&
       resource.data.uid == request.auth.uid;

  allow update:
    if false;

  allow delete:
    if signedIn() &&
       resource.data.uid == request.auth.uid;

}

match /reviews/{reviewId} {

  allow create:
    if signedIn();

  allow read:
    if true;

  allow update:
    if isAdmin();

  allow delete:
    if isAdmin();

}

match /productQuestions/{questionId} {

  allow create:
    if signedIn();

  allow read:
    if true;

  allow update:
    if isAdmin();

  allow delete:
    if isAdmin();

}

match /rewardHistory/{historyId} {

  allow create:
    if signedIn();

  allow read:
    if isAdmin() ||
       resource.data.uid == request.auth.uid;

  allow update,
        delete:
    if false;

}

  }
}
